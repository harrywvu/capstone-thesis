\begin{algorithm}
	\caption{SEKM Rekeying Algorithm - $O(N)$}
	\label{SEKMRekeyingAlgorithm}
	\begin{algorithmic}[1]
		\State Select $k-1$ degree polynomial.
		\State Broadcast witness polynomials and its hashed signature to the server group.
		\State For each $j$ server in the server group.
		\State Compute each $S_{i \rightarrow j}$ and send it together with its hashed signature to the server group.
		\State For each server $i$ in the server group.
		\State Compute new share ${S_j}^\textrm' = S_j + S_{i \rightarrow j}$.
		\State Replace the old share with the new share.
	\end{algorithmic}
\end{algorithm}
